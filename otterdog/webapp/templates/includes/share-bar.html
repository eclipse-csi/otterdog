{# The split of a whole into parts, e.g. the repositories by their secret scanning, as a segmented bar styled by css/app.scss (otterdog-share): each part a block as wide as its share in its status color, the share printed in the blocks wide enough for it, every part named in the legend, its count and share in the tooltip of its block and in a text for screen readers, the bar itself being hidden from them. name: the name of the whole, for screen readers parts: a list of {label, count, status}, in the order of the bar, status being good, warning or critical #} {% macro share_bar(name, parts) %} {% set total = parts|sum(attribute="count") %}

{{ name }}: {% for part in parts %}{{ part.label }} {{ part.count }}{{ '' if loop.last else ', ' }}{% endfor %}

{% for part in parts if part.count > 0 %} {% set share = part.count / total * 100 %}
{% if share >= 10 %}{{ '%.0f' % share }}%{% endif %}
{% else %}
{% endfor %}
    {% for part in parts %}
  • {{ part.label }}
    • {% endfor %}
